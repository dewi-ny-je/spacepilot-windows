cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build, for example: cmake --preset x64")
endif()
set(AXIAL_VERSION "0.2.4" CACHE STRING "Product version; release CI supplies the tag version")
if(NOT AXIAL_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "AXIAL_VERSION must be MAJOR.MINOR.PATCH")
endif()
project(Axial VERSION ${AXIAL_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# The settings app's model tests load the device catalog on any host.
if(NOT WIN32)
  add_library(axial-bridge SHARED src/catalog.cpp)
  target_include_directories(axial-bridge PRIVATE include)
  set_target_properties(axial-bridge PROPERTIES CXX_VISIBILITY_PRESET hidden)
  return()
endif()

# DLLs are loaded into third-party processes; link the C and C++ runtimes
# statically so they never depend on a redistributable being installed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
option(AXIAL_ADAPTERS_ONLY "Build only siappdll.dll and TDxNavLib.dll (32-bit packages)" OFF)

add_library(axial-options INTERFACE)
target_include_directories(axial-options INTERFACE include third_party/freecad)
target_compile_definitions(axial-options INTERFACE _WIN32_WINNT=0x0A00 WINVER=0x0A00 _CRT_SECURE_NO_WARNINGS)
target_link_libraries(axial-options INTERFACE ws2_32)
if(MSVC)
  target_compile_options(axial-options INTERFACE /W3 /permissive- /utf-8 /EHsc /Zc:__cplusplus /bigobj /wd4996)
else()
  target_compile_options(axial-options INTERFACE -Wall -Wextra -Wno-unused-parameter -Wno-deprecated-declarations -Wno-cast-function-type -Wno-missing-field-initializers -Wno-misleading-indentation)
  target_link_options(axial-options INTERFACE -static -static-libgcc -static-libstdc++)
endif()

include(cmake/WebDependencies.cmake)

# Adapters: names and exports match the 3DxWare libraries applications load.
add_library(siappdll SHARED src/siapp.cpp src/siappdll.def)
target_link_libraries(siappdll PRIVATE axial-options axial-boost)
add_library(TDxNavLib SHARED src/navlib.cpp src/TDxNavLib.def)
target_link_libraries(TDxNavLib PRIVATE axial-options axial-boost dwmapi user32)
set_target_properties(siappdll TDxNavLib PROPERTIES PREFIX "" IMPORT_PREFIX "")
if(AXIAL_ADAPTERS_ONLY)
  return()
endif()

add_library(axial-web STATIC src/web.cpp)
target_link_libraries(axial-web PUBLIC axial-options axial-web-dependencies)
add_executable(axial-web-setup src/web_setup.cpp)
target_link_libraries(axial-web-setup PRIVATE axial-options axial-web-dependencies crypt32)
add_executable(axial-service src/service.cpp)
target_link_libraries(axial-service PRIVATE axial-web hid setupapi advapi32 user32 psapi)
add_executable(axialctl src/ctl.cpp)
target_link_libraries(axialctl PRIVATE axial-options)
add_library(axial-bridge SHARED src/bridge.cpp src/catalog.cpp)
target_link_libraries(axial-bridge PRIVATE axial-options)
set_target_properties(axial-bridge PROPERTIES PREFIX "")
if(NOT MSVC)
  target_link_options(axial-web-setup PRIVATE -municode)
endif()

if(BUILD_TESTING)
  include(cmake/Tests.cmake)
endif()
